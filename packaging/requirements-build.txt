# Dependencies for BUILDING the application (frozen). Installed in CI/locally before PyInstaller.
# The runtime graph is intentionally narrow: numpy/pandas/matplotlib(Agg)/tkinter + aiosonic/orjson/onecache
# (Binance data fetcher). scipy/seaborn/statsmodels are NOT in the bundle (excluded in .spec).
numpy>=1.24,<2.3  # pinned to the numba-validated line: CI legs floated to 2.4.x while
                  # dev ran 2.2.x — three platforms must freeze ONE tested pair
pandas>=2.0,<3    # pin to 2.x: pandas 3's Copy-on-Write returns read-only .to_numpy() arrays; the
                  # code handles it, but this keeps the build matching the validated 2.x bundle
pytz              # quantpylib.wrappers.binance imports it directly (pandas 2.x no longer guarantees it)
numba             # JIT-compiles the fast_sim fitness kernel (~4x node); optional at runtime (numpy fallback)
customtkinter     # the GUI toolkit. NOT optional: alphanode_gui imports it at module level.
                  # Its assets (themes/*.json, fonts) are data files — pyinstaller-hooks-contrib
                  # ships hook-customtkinter.py, which collects them; --role selfcheck proves it.
darkdetect        # customtkinter dep: reads the OS light/dark preference for the first run
matplotlib>=3.6,<3.11  # 3.11.1 wheel landed without widgets.py on 2026-08-29: backend_bases died at import; stay on the validated 3.10 line
requests          # quantpylib.throttler.aiosonic imports requests
cryptography      # vault.py (formula sealing: x25519 + ChaCha20Poly1305) — node.py imports it
                  # whenever the shipped vault_server_key.pub exists, i.e. in every real build
aiosonic==0.31.1  # quantpylib is verified on 0.31.x (the API changed in 1.x)
orjson
onecache
pillow            # icon generation (make_icon.py)
pyinstaller>=6.0
cython>=3.0       # compiles the engine core to native extensions (cythonize_engine.py) — the
setuptools        # .spec refuses to ship the engine as decompilable bytecode without them
cryptography      # vault sealing (X25519+ChaCha20). The node seals EVERY mined formula, so this
                  # is a hard runtime dep of the frozen bundle — v1.4.1 shipped without it and
                  # the first search round died on `import vault` (selfcheck now proves the seal)
